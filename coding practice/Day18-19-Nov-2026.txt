# ANSIBLE — CONFIGURE HAPROXY FOR LOAD BALANCING

---
- name: Setup HAProxy Load Balancer
  hosts: all
  become: yes

  tasks:
    - name: Install HAProxy
      yum:
        name: haproxy
        state: present

    - name: Configure HAProxy
      copy:
        dest: /etc/haproxy/haproxy.cfg
        content: |
          global
              log /dev/log local0
              maxconn 2000

          defaults
              mode http
              timeout connect 5s
              timeout client  50s
              timeout server  50s

          frontend http_front
              bind *:80
              default_backend web_servers

          backend web_servers
              balance roundrobin
              server web1 10.0.1.10:80 check
              server web2 10.0.2.10:80 check

    - name: Validate HAProxy configuration
      command: haproxy -c -f /etc/haproxy/haproxy.cfg
      changed_when: false

    - name: Start and enable HAProxy
      service:
        name: haproxy
        state: restarted
        enabled: yes


# TERRAFORM — AUTO-SCALING GROUP WITH ALB

resource "aws_lb" "pathnex_lb" {
  name               = "pathnex-lb"
  internal           = false
  load_balancer_type = "application"

  security_groups = [
    aws_security_group.allow_ssh_http.id
  ]

  subnets = [
    aws_subnet.public.id,
    aws_subnet.public2.id
  ]
}

resource "aws_lb_target_group" "pathnex_tg" {
  name     = "pathnex-target-group"
  port     = 80
  protocol = "HTTP"
  vpc_id   = aws_vpc.main.id

  health_check {
    path = "/"
    port = "traffic-port"
  }
}

resource "aws_launch_template" "pathnex_config" {
  name_prefix   = "pathnex-template"
  image_id      = "ami-REPLACE_WITH_VALID_AMI"
  instance_type = "t2.micro"

  vpc_security_group_ids = [
    aws_security_group.allow_ssh_http.id
  ]

  user_data = base64encode(<<-EOF
    #!/bin/bash
    yum install -y httpd
    systemctl enable httpd
    systemctl start httpd
    echo "Pathnex Web Server" > /var/www/html/index.html
  EOF
  )
}

resource "aws_autoscaling_group" "pathnex_asg" {
  name                = "pathnex-asg"
  desired_capacity    = 3
  max_size            = 5
  min_size            = 2

  vpc_zone_identifier = [
    aws_subnet.public.id,
    aws_subnet.public2.id
  ]

  target_group_arns = [
    aws_lb_target_group.pathnex_tg.arn
  ]

  health_check_type         = "ELB"
  health_check_grace_period = 300

  launch_template {
    id      = aws_launch_template.pathnex_config.id
    version = "$Latest"
  }

  tag {
    key                 = "Name"
    value               = "Pathnex-ASG-EC2"
    propagate_at_launch = true
  }
}

resource "aws_lb_listener" "pathnex_listener" {
  load_balancer_arn = aws_lb.pathnex_lb.arn
  port              = 80
  protocol          = "HTTP"

  default_action {
    type             = "forward"
    target_group_arn = aws_lb_target_group.pathnex_tg.arn
  }
}


# KUBERNETES — HORIZONTAL POD AUTOSCALER

apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-web-app

spec:
  replicas: 3

  selector:
    matchLabels:
      app: pathnex-web-app

  template:
    metadata:
      labels:
        app: pathnex-web-app

    spec:
      containers:
        - name: nginx
          image: nginx:latest

          ports:
            - containerPort: 80

          resources:
            requests:
              cpu: 100m
              memory: 128Mi
            limits:
              cpu: 500m
              memory: 256Mi

---
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-web-app-hpa

spec:
  scaleTargetRef:
    apiVersion: apps/v1

  minReplicas: 2
  maxReplicas: 10

  metrics:
    - type: Resource
      resource:
        name: cpu
        target:
          type: Utilization

            steps {
                script {
                    sh '''
                    kubectl apply -f kubernetes/blue-deployment.yaml

                    kubectl rollout status deployment/blue-deployment

                    kubectl apply -f kubernetes/green-deployment.yaml

                    kubectl rollout status deployment/green-deployment

                    kubectl apply -f kubernetes/active-service.yaml
                    '''
                }
            }
        }
    }

    post {
        success {
            echo 'Blue-Green deployment completed!'
        }

        failure {
            echo 'Deployment failed!'
        }
    }
}


# GITLAB CI/CD — BLUE-GREEN DEPLOYMENT

stages:
  - build
  - push
  - deploy

variables:
  IMAGE_NAME: "$CI_REGISTRY_IMAGE:$CI_COMMIT_SHORT_SHA"

build:
  stage: build
  script:
    - docker build -t "$IMAGE_NAME" .

push:
  stage: push
  script:
    - echo "$CI_REGISTRY_PASSWORD" | docker login "$CI_REGISTRY" -u "$CI_REGISTRY_USER" --password-stdin
    - docker push "$IMAGE_NAME"

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/blue-deployment.yaml
    - kubectl rollout status deployment/blue-deployment
    - kubectl apply -f kubernetes/green-deployment.yaml
    - kubectl rollout status deployment/green-deployment
    - kubectl apply -f kubernetes/active-service.yaml
