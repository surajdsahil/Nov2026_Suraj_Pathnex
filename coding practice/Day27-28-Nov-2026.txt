
# ANSIBLE — DEPLOY SERVERLESS AWS LAMBDA FUNCTION

- name: Deploy Serverless Lambda Function
  hosts: localhost
  connection: local
  gather_facts: no

  tasks:
    - name: Create Lambda deployment package
      command: zip -r lambda.zip lambda/
      args:
        chdir: "{{ playbook_dir }}"

    - name: Deploy Lambda function
      command: >
        aws lambda create-function
        --function-name pathnex-lambda
        --runtime nodejs22.x
        --role arn:aws:iam::123456789012:role/lambda-role
        --handler index.handler
        --zip-file fileb://lambda.zip
      args:
        chdir: "{{ playbook_dir }}"
      register: lambda_result
      failed_when:
        - lambda_result.rc != 0
        - "'ResourceConflictException' not in lambda_result.stderr"

    - name: Update Lambda code if function already exists
      command: >
        aws lambda update-function-code
        --function-name pathnex-lambda
        --zip-file fileb://lambda.zip
      args:
        chdir: "{{ playbook_dir }}"
      when: "'ResourceConflictException' in lambda_result.stderr"

  name: pathnex-lambda
  namespace: default

  runtime: nodejs18
  handler: index.handler
  function: |
    module.exports = {
      handler: async function(event, context) {
        return {
          statusCode: 200,
          body: "Hello Pathnex"
        };
      }
    };


# JENKINSFILE — LAMBDA FUNCTION DEPLOYMENT PIPELINE

pipeline {
    agent any

    environment {
        FUNCTION_NAME = 'pathnex-lambda'
        AWS_DEFAULT_REGION = 'us-east-1'
    }

    stages {
        stage('Build') {
            steps {
                echo 'Building Lambda function package...'
                sh 'cd lambda && zip -r ../lambda.zip .'
            }
        }

        stage('Deploy to Lambda') {
            steps {
                sh '''
                    aws lambda update-function-code \
                      --function-name "$FUNCTION_NAME" \
                      --zip-file fileb://lambda.zip

                    aws lambda wait function-updated \
                      --function-name "$FUNCTION_NAME"
                '''
            }
        }
    }
}
