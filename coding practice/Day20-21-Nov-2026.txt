
# ANSIBLE — CONFIGURE FIREWALL WITH IPTABLES

- name: Configure Firewall with iptables
  hosts: all
  become: yes

  tasks:
    - name: Allow established connections
      iptables:
        chain: INPUT
        ctstate:
          - ESTABLISHED
          - RELATED
        jump: ACCEPT

    - name: Allow loopback traffic
      iptables:
        chain: INPUT
        in_interface: lo
        jump: ACCEPT

    - name: Allow SSH traffic
      iptables:
        chain: INPUT
        protocol: tcp
        destination_port: 22
        jump: ACCEPT

    - name: Allow HTTP traffic
      iptables:
        chain: INPUT
        protocol: tcp
        destination_port: 80
        jump: ACCEPT

    - name: Allow HTTPS traffic
      iptables:
        chain: INPUT
        protocol: tcp
        destination_port: 443
        jump: ACCEPT

    - name: Set default INPUT policy to DROP
      iptables:
        chain: INPUT
        policy: DROP

    - name: Allow outgoing traffic
      iptables:
        chain: OUTPUT
        policy: ACCEPT

    - name: Allow forwarded traffic
      iptables:
        chain: FORWARD
        policy: DROP


# TERRAFORM — CONFIGURE AWS SECURITY GROUP

resource "aws_security_group" "pathnex_sg" {
  name        = "pathnex-security-group"
  description = "Allow SSH, HTTP and HTTPS"
  vpc_id      = aws_vpc.main.id

  ingress {
    description = "SSH access - restrict to trusted IP"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["YOUR_PUBLIC_IP/32"]
  }

  ingress {
    description = "HTTP access"
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  ingress {
    description = "HTTPS access"
    from_port   = 443
    to_port     = 443
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    description = "Allow all outbound traffic"
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }

  tags = {
    Name = "pathnex-security-group"
  }
}


# KUBERNETES — RBAC CONFIGURATION

apiVersion: v1
kind: ServiceAccount
metadata:
  name: pathnex-user
  namespace: default
---
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: pathnex-role
  namespace: default
rules:
  - apiGroups: [""]
    resources: ["pods", "services"]
    verbs: ["get", "list", "watch"]
  - apiGroups: ["apps"]
    resources: ["deployments"]
    verbs: ["get", "list", "watch"]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: pathnex-role-binding
  namespace: default
subjects:
  - kind: ServiceAccount
    name: pathnex-user
    namespace: default
roleRef:
  kind: Role
  name: pathnex-role
  apiGroup: rbac.authorization.k8s.io


# JENKINSFILE — SECURE CREDENTIALS MANAGEMENT

pipeline {
    agent any

    environment {
        IMAGE_NAME = 'pathnex-web-app'
        REGISTRY_CREDENTIALS = 'docker-credentials'
    }

    stages {
        stage('Build') {
            steps {
                script {
                    def appImage = docker.build(IMAGE_NAME)
                }
            }
        }

        stage('Push') {
            steps {
                script {
                    docker.withRegistry('https://index.docker.io/v1/', REGISTRY_CREDENTIALS) {
                        docker.image(IMAGE_NAME).push('latest')
                    }
                }
            }
        }
    }
}


# GITLAB CI/CD — SECURE DEPLOYMENT

stages:
  - build
  - push
  - deploy

variables:
  IMAGE_TAG: "$CI_REGISTRY_IMAGE:$CI_COMMIT_SHORT_SHA"

build:
  stage: build
  script:
    - docker build -t "$IMAGE_TAG" .

push:
  stage: push
  script:
    - echo "$CI_REGISTRY_PASSWORD" | docker login "$CI_REGISTRY" -u "$CI_REGISTRY_USER" --password-stdin
    - docker push "$IMAGE_TAG"

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml
    - kubectl rollout status deployment/pathnex-web-app --timeout=120s
    - kubectl get pods
  environment:
    name: production


# DOCKER — NETWORKING

# Create a custom bridge network
docker network create pathnex-network

# Run first container
docker run -dit --name c1 --network pathnex-network ubuntu:24.04

# Run second container
docker run -dit --name c2 --network pathnex-network ubuntu:24.04

# Check running containers
docker ps

# Inspect network
docker network inspect pathnex-network

# Test communication between containers
docker exec c1 bash -c "apt-get update && apt-get install -y iputils-ping"
docker exec c1 ping -c 4 c2

# Open shell inside containers
docker exec -it c1 bash
docker exec -it c2 bash
